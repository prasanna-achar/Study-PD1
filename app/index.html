Best-Practice Trigger Handler Skeleton

trigger AccountTrigger on Account (before insert, before update, after insert, after update) {
    AccountTriggerHandler.run();
}

public with sharing class AccountTriggerHandler {
    public static void run() {
        if (Trigger.isBefore && Trigger.isUpdate) {
            onBeforeUpdate((List<Account>)Trigger.new, (Map<Id, Account>)Trigger.oldMap);
        }
    }
    
    private static void onBeforeUpdate(List<Account> newAccs, Map<Id, Account> oldMap) {
        for (Account acc : newAccs) {
            // Compare new vs old value using oldMap to process only when changed!
            if (acc.Rating == 'Hot' && oldMap.get(acc.Id).Rating != 'Hot') {
                acc.Description = 'Upgraded to Hot Account priority.';
            }
        }
    }
}

LWC @wire Adapter vs Imperative Apex

// Apex Controller: Must use @AuraEnabled(cacheable=true) for @wire!
public with sharing class AccountControllerLwc {
    @AuraEnabled(cacheable=true)
    public static List<Account> getActiveAccounts(String industry) {
        return [SELECT Id, Name FROM Account WHERE Industry = :industry];
    }
}

// LWC JavaScript (accountViewer.js)
import { LightningElement, api, wire } from 'lwc';
import getActiveAccounts from '@salesforce/apex/AccountControllerLwc.getActiveAccounts';

export default class AccountViewer extends LightningElement {
    @api recordId;
    @api industryFilter = 'Technology';

    // Wired Apex property
    @wire(getActiveAccounts, { industry: '$industryFilter' })
    accounts;
}

Comprehensive @isTest Pattern with TestSetup

@isTest
private class ComprehensiveTests {
    @TestSetup
    static void setupData() {
        insert new Account(Name = 'Test Setup Account', Active__c = 'Yes');
    }

    @isTest
    static void testGovernorLimitReset() {
        List<Account> accs = [SELECT Id FROM Account];
        
        Test.startTest(); // Gives a fresh set of 100 SOQL / 150 DML limits!
        accs[0].AnnualRevenue = 500000;
        update accs;
        Test.stopTest(); // Forces all async code (@future, Batchable) to complete synchronously!

        Account updated = [SELECT AnnualRevenue FROM Account WHERE Id = :accs[0].Id];
        Assert.areEqual(500000, updated.AnnualRevenue, 'Revenue should match.');
    }
}
